## 1. Local Database & Storage Architecture

To achieve zero-latency performance, minimal RAM footprint (<30 MB overhead), and full offline capability without heavy external database servers, Aegis-Context uses a dual-engine embedded architecture powered by **SQLite** (using the ultra-efficient `sqlite-vector` extension for BLOB-based vector storage) coupled with a rolling transactional store.

### File System Layout

```text
~/.aegis/
├── aegis_state.db        # Main SQLite instance (Hot ring buffer & metadata)
├── aegis_vectors.db      # Long-term Warm Store + Vector Embeddings
└── config.json           # Runtime parameters (capture interval, retention policies)

```

## 2. Comprehensive Database Schema (`aegis_vectors.db`)

This schema handles long-term storage, metadata tagging (active app, window title, timestamp), and raw embedding storage without requiring external vector database daemons.

```sql
-- Enable foreign key constraints and write-ahead logging for high concurrency
PRAGMA foreign_keys = ON;
PRAGMA journal_mode = WAL;

-- 1. Applications Registry Table
CREATE TABLE IF NOT EXISTS applications (
    app_id INTEGER PRIMARY KEY AUTOINCREMENT,
    bundle_identifier TEXT UNIQUE NOT NULL, -- e.g., 'com.microsoft.vscode', 'com.google.chrome'
    display_name TEXT NOT NULL,             -- e.g., 'Visual Studio Code', 'Google Chrome'
    os_platform TEXT NOT NULL               -- 'windows', 'macos', 'android'
);

-- 2. Master Context Sessions (Window / Screen State blocks)
CREATE TABLE IF NOT EXISTS screen_contexts (
    context_id INTEGER PRIMARY KEY AUTOINCREMENT,
    app_id INTEGER NOT NULL,
    window_title TEXT,
    raw_ocr_text TEXT NOT NULL,             -- Extracted text from VLM/OCR engine
    semantic_summary TEXT NOT NULL,         -- Dense summary generated by Moondream2
    captured_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    embedding_blob BLOB,                    -- Serialized Float32 vector array (e.g., 384-dim or 768-dim)
    FOREIGN KEY (app_id) REFERENCES applications(app_id) ON DELETE CASCADE
);

-- 3. Create indices for instant temporal filtering and application-based queries
CREATE INDEX IF NOT EXISTS idx_contexts_timestamp ON screen_contexts(captured_at DESC);
CREATE INDEX IF NOT EXISTS idx_contexts_app ON screen_contexts(app_id);

-- 4. Tier-2 Summarization Ledger (For rolling rollups of older context)
CREATE TABLE IF NOT EXISTS historical_digests (
    digest_id INTEGER PRIMARY KEY AUTOINCREMENT,
    start_time DATETIME NOT NULL,
    end_time DATETIME NOT NULL,
    macro_summary TEXT NOT NULL,            -- Summarized narrative of what the user did over a larger time block (e.g., 1 hour)
    digest_embedding BLOB                   -- Vector representation of the macro summary
);

```

## 3. Python Backend Engine Implementation (`storage_engine.py`)

This module manages the local SQLite connection, handles direct BLOB vector inserts using `sqlite-vector`, and processes Tier-1 to Tier-2 background memory rollups.

```python
import sqlite3
import struct
import numpy as np
import importlib.resources

class AegisStorageEngine:
    def __init__(self, db_path="aegis_vectors.db"):
        self.db_path = db_path
        self.conn = sqlite3.connect(self.db_path, check_same_thread=False)
        self._initialize_extension_and_schema()

    def _initialize_extension_and_schema(self):
        # Load native sqlite-vector extension binaries safely
        try:
            ext_path = importlib.resources.files("sqlite_vector.binaries") / "vector"
            self.conn.enable_load_extension(True)
            self.conn.load_extension(str(ext_path))
        except Exception:
            # Fallback for manual local binary compilation or standard fallback execution
            pass
        
        cursor = self.conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS applications (
                app_id INTEGER PRIMARY KEY AUTOINCREMENT,
                bundle_identifier TEXT UNIQUE NOT NULL,
                display_name TEXT NOT NULL,
                os_platform TEXT NOT NULL
            );
        """)
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS screen_contexts (
                context_id INTEGER PRIMARY KEY AUTOINCREMENT,
                app_id INTEGER NOT NULL,
                window_title TEXT,
                raw_ocr_text TEXT NOT NULL,
                semantic_summary TEXT NOT NULL,
                captured_at DATETIME DEFAULT CURRENT_TIMESTAMP,
                embedding_blob BLOB,
                FOREIGN KEY (app_id) REFERENCES applications(app_id)
            );
        """)
        self.conn.commit()

    def serialize_vector(self, vector: np.ndarray) -> bytes:
        """Convert a numpy float32 vector array into a raw BLOB for SQLite storage."""
        return vector.astype(np.float32).tobytes()

    def deserialize_vector(self, blob: bytes) -> np.ndarray:
        """Unpack a raw BLOB back into a numpy float32 array."""
        return np.frombuffer(blob, dtype=np.float32)

    def log_screen_state(self, bundle_id: str, app_name: str, platform: str, window_title: str, ocr_text: str, summary: str, embedding: np.ndarray):
        cursor = self.conn.cursor()
        
        # Upsert application entry
        cursor.execute("""
            INSERT INTO applications (bundle_identifier, display_name, os_platform)
            VALUES (?, ?, ?)
            ON CONFLICT(bundle_identifier) DO UPDATE SET display_name=excluded.display_name;
        """, (bundle_id, app_name, platform))
        
        cursor.execute("SELECT app_id FROM applications WHERE bundle_identifier = ?", (bundle_id,))
        app_id = cursor.fetchone()[0]

        # Insert screen state context block
        blob_data = self.serialize_vector(embedding) if embedding is not None else None
        cursor.execute("""
            INSERT INTO screen_contexts (app_id, window_title, raw_ocr_text, semantic_summary, embedding_blob)
            VALUES (?, ?, ?, ?, ?)
        """, (app_id, window_title, ocr_text, summary, blob_data))
        
        self.conn.commit()

    def vector_similarity_search(self, query_vector: np.ndarray, top_k: int = 5):
        """Perform high-performance local vector similarity search utilizing optimized distance scans."""
        cursor = self.conn.cursor()
        cursor.execute("SELECT context_id, semantic_summary, captured_at, embedding_blob FROM screen_contexts WHERE embedding_blob IS NOT NULL")
        rows = cursor.fetchall()

        results = []
        q_vec = query_vector.astype(np.float32)
        
        for row in rows:
            ctx_id, summary, timestamp, blob = row
            db_vec = self.deserialize_vector(blob)
            
            # Compute Cosine Similarity locally using numpy (optimized via SIMD BLAS backends)
            similarity = np.dot(q_vec, db_vec) / (np.linalg.norm(q_vec) * np.linalg.norm(db_vec) + 1e-10)
            results.append((similarity, ctx_id, summary, timestamp))

        # Sort descending by highest similarity score
        results.sort(key=lambda x: x[0], reverse=True)
        return results[:top_k]

```

## 4. End-to-End Execution Flow & Pipeline Management

1. **Capture Daemon Trigger:** Fires every 5 seconds (adaptive based on user activity). Captures frame buffer using cross-platform UI display hooks (`mss` or platform primitives).
2. **Vision Extraction:** Compresses image, sends to local **Moondream2** model instance running via `llama.cpp` / Ollama, yielding structural Markdown and OCR text.
3. **Embedding Generation:** Passes text output through a ultra-lightweight local embedding model (e.g., `all-MiniLM-L6-v2` compiled to ONNX, running in $<10\text{ms}$ on CPU).
4. **Storage Commit:** Writes the structured data and vector blob securely to `aegis_vectors.db`.
5. **Context Window Assembly:** When a user executes the global hotkey shortcut, the query engine extracts the immediate Tier-1 RAM circular ring buffer combined with a vector similarity search across Tier-2 SQLite database, injecting the exact needed historical state directly into the local model context without a single byte leaving the device.

## 1. Executive Summary & Product Objective

Traditional software requires explicit intent: opening an app, navigating menus, and typing queries. This breaks cognitive flow and imposes a heavy context-switching penalty.

The objective of Module II is to eliminate manual invocation friction entirely. By connecting the local ambient memory daemon (from PRD Part 1) to native operating system layers (Android AppFunctions / iOS App Intents, local loopback APIs, and desktop global hotkeys), users can query their past workspace state via voice (*"Hey Google / Hey Siri, what was I looking at 5 minutes ago?"*) or a single keystroke (`Ctrl+Space`) **without ever leaving their current application viewport or unlocking their screen UI.**

## 3. Core Component Specifications

### 3.1 Local Loopback IPC Server (`ipc_server.py`)

* **Function:** Acts as the bridge between operating system shortcuts, mobile assistants, desktop HUDs, and the background SQLite database/RAM buffer.
* **Technology:** FastAPI running asynchronously via Uvicorn, bound exclusively to loopback (`127.0.0.1:8765`) to ensure zero external network exposure.
* **Endpoints:**
* `GET /context/recent`: Returns the immediate Tier-1 RAM circular ring buffer (last 5 minutes of activity).
* `POST /context/query`: Accepts a natural language query string, runs vector similarity search against `aegis_vectors.db`, and returns a synthesized summary.



### 3.2 Mobile Voice & System Assistant Integration (Android/iOS)

* **Android Implementation:** Leverages modern Android **AppFunctions** / App Actions infrastructure to expose local app data directly to system-level voice models.
* A background `ContentProvider` or exported local service exposes the recent state to Google Assistant. When a user prompts *"Hey Google, what was I doing?"*, the system assistant queries `[http://127.0.0.1:8765/context/recent](http://127.0.0.1:8765/context/recent)`, reads the JSON payload, and speaks the summary aloud via native TTS synthesis without launching the application GUI.


* **iOS Implementation:** Implements local **App Intents** framework linked to background app groups, allowing Siri Shortcuts to query local context seamlessly.

### 3.3 Desktop Floating HUD & Global Hotkey Interface

* **Function:** Eliminates context switching during intense coding or reading sessions by overlaying a minimalist, non-intrusive floating heads-up display.
* **Behavior:**
1. Triggered via global shortcut (`Ctrl+Shift+Space` or `Cmd+Shift+Space`).
2. Surfaces a borderless, dark-themed text box directly beneath the active cursor.
3. Displays a single-line real-time summary of the last 2 minutes of workspace activity.
4. Pressing `Enter` instantly injects the structured context string straight into the system clipboard, ready to paste into Cursor, VS Code, or browser prompts, or streams an inline answer from the local LLM.

## 4. Implementation Code Blueprints

### 4.1 Local Loopback IPC Server Module (`ipc_server.py`)

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import uvicorn
import numpy as np

app = FastAPI(title="Aegis Ambient IPC Bridge", version="1.0.0")

# Reference to the running storage engine and ring buffer from Part 1 daemon
class ContextBridgeState:
    storage_engine = None
    ring_buffer = []

state = ContextBridgeState()

class QueryRequest(BaseModel):
    query_text: str
    top_k: int = 3

@app.get("/context/recent")
def get_recent_context():
    """Returns the immediate Tier-1 hot RAM buffer for instant voice/system recall."""
    if not state.ring_buffer:
        return {"status": "ok", "recent_context": "No recent activity recorded."}
    return {
        "status": "ok",
        "count": len(state.ring_buffer),
        "recent_context": state.ring_buffer[-5:] # Return last 5 states
    }

@app.post("/context/query")
def query_historical_context(payload: QueryRequest):
    """Performs semantic similarity lookup against Tier-2 warm storage."""
    if not state.storage_engine:
        raise HTTPException(status_code=500, status.HTTP_500_INTERNAL_SERVER_ERROR, detail="Storage engine not bound.")
    
    # In production: convert payload.query_text to embedding via ONNX model, then search
    dummy_query_vector = np.random.randn(384).astype(np.float32)
    results = state.storage_engine.similarity_search(dummy_query_vector, top_k=payload.top_k)
    
    formatted_results = [
        {"similarity_score": float(res[0]), "window_title": res[2], "summary": res[3], "timestamp": res[4]}
        for res in results
    ]
    return {"status": "ok", "matches": formatted_results}

def start_ipc_server(storage_engine, ring_buffer_ref, port=8765):
    state.storage_engine = storage_engine
    state.ring_buffer = ring_buffer_ref
    uvicorn.run(app, host="127.0.0.1", port=port, log_level="warning")

```

### 4.2 Desktop Floating HUD & Hotkey Listener Daemon (`hud_interface.py`)

```python
import sys
import threading
import requests
from PyQt5.QtWidgets import QApplication, QWidget, QVBoxLayout, QLabel, QLineEdit
from PyQt5.QtCore import Qt, QTimer
from PyQt5.QtGui import QFont

class AegisFloatingHUD(QWidget):
    def __init__(self):
        super().__init__()
        self.init_ui()

    def init_ui(self):
        # Window flags for transparent, borderless, always-on-top HUD overlay
        self.setWindowFlags(Qt.FramelessWindowHint | Qt.WindowStaysOnTopHint | Qt.Tool)
        self.setAttribute(Qt.WA_TranslucentBackground, True)
        self.resize(600, 120)

        layout = QVBoxLayout()
        layout.setContentsMargins(15, 15, 15, 15)

        # Style container for sleek cyberpunk/minimalist developer aesthetic
        self.setStyleSheet("""
            QWidget {
                background-color: rgba(20, 20, 30, 235);
                border: 1px solid rgba(100, 100, 255, 100);
                border-radius: 10px;
                color: #FFFFFF;
            }
            QLineEdit {
                background-color: rgba(30, 30, 45, 255);
                border: 1px solid #444;
                border-radius: 5px;
                padding: 8px;
                color: #00FFCC;
                font-family: 'Courier New', monospace;
            }
        """)

        self.label = QLabel("Aegis Ambient Context: Initializing...", self)
        self.label.setFont(QFont("Arial", 10))
        layout.addWidget(self.label)

        self.input_field = QLineEdit(self)
        self.input_field.setPlaceholderText("Ask what you were doing or hit Enter to copy context...")
        self.input_field.returnPressed.connect(self.handle_submission)
        layout.addWidget(self.input_field)

        self.setLayout(layout)
        self.center_on_screen()
        self.fetch_recent_context_preview()

    def center_on_screen(self):
        screen = QApplication.primaryScreen().geometry()
        x = (screen.width() - self.width()) // 2
        y = int(screen.height() * 0.25) # Position upper-middle viewport
        self.move(x, y)

    def fetch_recent_context_preview(self):
        try:
            res = requests.get("http://127.0.0.1:8765/context/recent", timeout=1)
            if res.status_code == 200:
                data = res.json().get("recent_context", [])
                if data:
                    latest = data[-1]
                    self.label.setText(f"Latest Context: {latest}")
                else:
                    self.label.setText("Latest Context: No active history.")
        except Exception:
            self.label.setText("Status: Daemon offline or unreachable.")

    def handle_submission(self):
        query = self.input_field.text()
        if not query:
            # Copy recent context to clipboard if input is empty
            clipboard = QApplication.clipboard()
            clipboard.setText(self.label.text())
            self.hide()
            return
        
        # Query local endpoint
        try:
            res = requests.post("http://127.0.0.1:8765/context/query", json={"query_text": query}, timeout=2)
            if res.status_code == 200:
                matches = res.json().get("matches", [])
                if matches:
                    self.label.setText(f"Match: {matches[0]['summary']}")
        except Exception:
            self.label.setText("Query failed: Local IPC error.")
        self.input_field.clear()

def run_hud_app():
    app = QApplication(sys.argv)
    hud = AegisFloatingHUD()
    hud.show()
    sys.exit(app.exec_())

```

## 5. Non-Functional Security & Performance Requirements

1. **Loopback Confinement:** The FastAPI IPC bridge binds *strictly* to `127.0.0.1` (`localhost`), ensuring that external network adapters cannot probe or sniff local workspace context packets.
2. **Sub-100ms IPC Latency:** Context retrieval requests from hotkeys or system assistants must resolve within 100 milliseconds to preserve the illusion of instant ambient recall.
3. **Zero-UI Footprint by Default:** The desktop HUD remains completely dormant in process memory, consuming 0% CPU until triggered by the global shortcut, keeping resource overhead minimal.
